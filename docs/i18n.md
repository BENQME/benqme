## Overview

Internationalization (i18n) enables the app to be presented in multiple languages and locales. This project uses **`next-intl`** for translation management and locale routing.

---

## Supported Locales

| Code | Language | Direction |
|------|----------|-----------|
| `en` | English (default) | LTR |
| `fr` | French | LTR |
| `ar` | Arabic (planned) | RTL |

---

## Setup

```bash
pnpm add next-intl
```

### Directory Structure
```
src/
├── i18n/
│   ├── request.ts          # Server-side locale resolution
│   └── routing.ts          # Locale routing config
├── messages/
│   ├── en.json             # English translations
│   └── fr.json             # French translations
├── app/
│   └── [locale]/           # Locale-scoped route group
│       ├── layout.tsx
│       └── page.tsx
└── middleware.ts            # Locale detection middleware
```

### Configuration
```typescript
// src/i18n/routing.ts
import { defineRouting } from "next-intl/routing";

export const routing = defineRouting({
  locales: ["en", "fr"],
  defaultLocale: "en",
  localePrefix: "as-needed", // /en/* only when not default
});
```

```typescript
// middleware.ts
import createMiddleware from "next-intl/middleware";
import { routing } from "./src/i18n/routing";

export default createMiddleware(routing);

export const config = {
  matcher: ["/((?!api|_next|_vercel|.*\\..*).*)"],
};
```

### Key Naming Conventions
- Keys use **camelCase**
- Organized by **feature namespace** (not type)
- Variables use **ICU message format**: `{variableName}`
- Keep keys short and descriptive

## Usage in Components

### Server Components
```tsx
import { getTranslations } from "next-intl/server";

export default async function HomePage() {
  const t = await getTranslations("navigation");

  return (
    <nav>
      <Link href="/">{t("home")}</Link>
      <Link href="/projects">{t("projects")}</Link>
    </nav>
  );
}
```

### Client Components
```tsx
"use client";

import { useTranslations } from "next-intl";

export function ErrorMessage() {
  const t = useTranslations("common");

  return (
    <div role="alert">
      <p>{t("error")}</p>
      <button onClick={retry}>{t("retry")}</button>
    </div>
  );
}
```

### Variables in Translations
```tsx
// Translation: "publishedOn": "Published on {date}"
t("blog.publishedOn", { date: formatDate(post.publishedAt) })

// Translation: "readingTime": "{minutes} min read"
t("blog.readingTime", { minutes: readingTime(post.content) })
```

## Pluralization

Use ICU message format for pluralization:

```json
{
  "blog": {
    "postCount": "{count, plural, =0 {No posts} =1 {1 post} other {# posts}}"
  }
}
```

```tsx
t("blog.postCount", { count: posts.length })
// 0 → "No posts"
// 1 → "1 post"
// 5 → "5 posts"
```

## Date, Number & Currency Formatting

Use `Intl` APIs with the user's locale for formatting:

```tsx
import { useLocale } from "next-intl";
import { formatDate, formatCurrency } from "@/utils/format";

export function PostMeta({ post }: { post: Post }) {
  const locale = useLocale();

  return (
    <time dateTime={post.publishedAt}>
      {new Intl.DateTimeFormat(locale, {
        year: "numeric",
        month: "long",
        day: "numeric",
      }).format(new Date(post.publishedAt))}
    </time>
  );
}
```

### Utility Wrappers
```typescript
// src/utils/format.ts
export function formatLocalDate(date: Date, locale: string): string {
  return new Intl.DateTimeFormat(locale, {
    year: "numeric",
    month: "long",
    day: "numeric",
  }).format(date);
}

export function formatLocalCurrency(amount: number, locale: string, currency: string): string {
  return new Intl.NumberFormat(locale, { style: "currency", currency }).format(amount);
}

export function formatLocalNumber(n: number, locale: string): string {
  return new Intl.NumberFormat(locale).format(n);
}
```

## RTL Support

For Arabic and other RTL languages:

```tsx
// src/app/[locale]/layout.tsx
import { getLocale } from "next-intl/server";

const RTL_LOCALES = ["ar", "he", "fa"];

export default async function LocaleLayout({ children, params }: Props) {
  const locale = await getLocale();
  const dir = RTL_LOCALES.includes(locale) ? "rtl" : "ltr";

  return (
    <html lang={locale} dir={dir}>
      <body>{children}</body>
    </html>
  );
}
```

### CSS for RTL
```css
/* Use logical properties instead of physical (left/right) */

/* ✅ Logical — works in both LTR and RTL */
.element {
  margin-inline-start: 1rem;
  padding-inline-end: 1rem;
  border-inline-start: 2px solid;
}

/* ❌ Physical — breaks in RTL */
.element {
  margin-left: 1rem;
  padding-right: 1rem;
  border-left: 2px solid;
}
```

## SEO & Metadata

```typescript
// src/app/[locale]/layout.tsx
import { getMessages } from "next-intl/server";
import type { Metadata } from "next";

export async function generateMetadata({ params: { locale } }: Props): Promise<Metadata> {
  const messages = await getMessages();

  return {
    title: {
      default: "BENQME",
      template: "%s | BENQME",
    },
    alternates: {
      canonical: `https://benqme.com/${locale}`,
      languages: {
        "en": "https://benqme.com",
        "fr": "https://benqme.com/fr",
        "x-default": "https://benqme.com",
      },
    },
  };
}
```

## Workflow

### Adding a New Translation Key
1. Add the key to `src/messages/en.json`
2. Add corresponding keys to all other locale files
3. Use the key in your component

### Adding a New Locale
1. Create `src/messages/{locale}.json`
2. Add locale to `routing.ts` locales array
3. Update `RTL_LOCALES` if needed
4. Add `hreflang` alternates in metadata

### Translation Review
- Machine translation is acceptable for initial strings
- All user-facing strings must be reviewed by a native speaker before shipping
- Untranslated strings fall back to the default locale (`en`)
